File Handling const dropzone = document.getElementById('dropzone'); const pdfInput = document.getElementById('pdf-input'); dropzone.onclick = () => pdfInput.click(); pdfInput.onchange = (e) => handleFile(e.target.files[0]); async function handleFile(file) { if (!file) return; const reader = new FileReader(); reader.onload = async function() { const typedarray = new Uint8Array(this.result); const loadingTask = pdfjsLib.getDocument(typedarray); const pdf = await loadingTask.promise; let fullText = ""; for (let i = 1; i <= pdf.numPages; i++) { const page = await pdf.getPage(i); const content = await page.getTextContent(); // CRITICAL FIX: Sort text fragments by Y then X coordinates // This puts the text back in "Reading Order" const sortedItems = content.items.sort((a, b) => { if (a.transform[5] !== b.transform[5]) { return b.transform[5] - a.transform[5]; // Top to bottom } return a.transform[4] - b.transform[4]; // Left to right }); fullText += sortedItems.map(item => item.str).join(" ") + "\n"; } parseText(fullText); }; reader.readAsArrayBuffer(file); } // 3. Extraction Logic (Date-Centric) function parseText(text) { const results = []; // Flexible regex: matches 2 digits, spaces, 3 Greek capitals, spaces, 2 digits const dateRegex = /(\d{2}\s+[Α-Ω]{3}\s+\d{2})/g; const dates = [...text.matchAll(dateRegex)]; dates.forEach((match, i) => { const dateStr = match[0]; const start = match.index; const end = dates[i+1] ? dates[i+1].index : text.length; const chunk = text.substring(start, end).toUpperCase(); // Identify the day const dayMatch = chunk.match(/ΔΕΥΤΕΡΑ|ΤΡΙΤΗ|ΤΕΤΑΡΤΗ|ΠΕΜΠΤΗ|ΠΑΡΑΣΚΕΥΗ|ΣΑΒΒΑΤΟ|ΚΥΡΙΑΚΗ/); const dayStr = dayMatch ? dayMatch[0] : "---"; const isWeekend = dayStr === "ΣΑΒΒΑΤΟ" || dayStr === "ΚΥΡΙΑΚΗ"; const isClosed = chunk.includes("ΚΛΕΙΣΤΑ"); results.push({ date: dateStr, day: dayStr, meal_a: isClosed ? "ΚΛΕΙΣΤΑ ΛΟΓΩ ΕΡΓΑΣΙΩΝ" : (isWeekend ? "Της Ώρας (Κατόπιν Παραγγελίας)" : "ΠΡΟΣ ΕΠΕΞΕΡΓΑΣΙΑ"), meal_b: "", status: isClosed ? "closed" : "open", type: isWeekend ? "weekend" : "weekday" }); }); updatePreview(results); } function updatePreview(data) { document.getElementById('json-output').value = JSON.stringify(data, null, 4); document.getElementById('day-count').innerText = `${data.length} Days Found`; document.getElementById('preview-section').classList.remove('hidden'); // Scroll to the preview area document.getElementById('preview-section').scrollIntoView({ behavior: 'smooth' }); } // 4. GitHub API Push async function pushToGitHub() { const token = localStorage.getItem('admin_gh_token'); const repo = localStorage.getItem('admin_gh_repo'); const btn = document.getElementById('update-btn'); const data = document.getElementById('json-output').value; if (!token || !repo) return alert("Please fill in Token and Repo first!"); try { btn.disabled = true; btn.innerText = "⏳ Committing to GitHub..."; const url = `https://api.github.com/repos/${repo}/contents/menu.json`; // We need to fetch the file first to get the 'sha' (GitHub requirement for updates) const getRes = await fetch(url, { headers: { 'Authorization': `token ${token}` } }); if (!getRes.ok) throw new Error("Could not find menu.json in the repo."); const getData = await getRes.json(); const payload = { message: `Admin Update: ${new Date().toLocaleDateString()}`, content: Base64.encode(data), sha: getData.sha }; const putRes = await fetch(url, { method: 'PUT', headers: { 'Authorization': `token ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (putRes.ok) { alert("🚀 Website Update Triggered!
